Hands-on Assignments for Displaying Data from Multiple Tables Using Joins

1.  

    SELECT EMPNO, ENAME, SAL, DNAME, LOC
    FROM EMP NATURAL JOIN DEPT;

2.  

    SELECT JOB, MGR, SAL, COMM, DNAME
    FROM EMP E JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO
    WHERE JOB='SALESMAN';

3.  

    SELECT ENAME, JOB, E.DEPTNO, DNAME
    FROM EMP E JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO
    WHERE LOC='DALLAS';

4.  

    SELECT E.ENAME "Employee", E.EMPNO "Emp#",
    M.ENAME "Manager", M.EMPNO "Mgr#"
    FROM EMP E JOIN EMP M
    ON E.MGR=M.EMPNO;

5.  

    SELECT E.ENAME "Employee", E.EMPNO "Emp#",
    M.ENAME "Manager", M.EMPNO "Mgr#"
    FROM EMP E LEFT OUTER JOIN EMP M
    ON E.MGR=M.EMPNO
    ORDER BY E.EMPNO;

6.  

    DESC SALGRADE;

    SELECT ENAME, JOB, DNAME, SAL, GRADE
    FROM EMP E JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO
    JOIN SALGRADE S
    ON E.SAL BETWEEN S.LOSAL AND S.HISAL;

7.  

    SELECT ENAME, DNAME
    FROM EMP E RIGHT OUTER JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO;

8.  

    SELECT E.ENAME, E.HIREDATE,
    M.ENAME MANAGER, M.HIREDATE
    FROM EMP E LEFT OUTER JOIN EMP M
    ON E.MGR=M.EMPNO
    WHERE E.HIREDATE<M.HIREDATE;

9.  

    SELECT EMPNO, ENAME, DNAME, LOC
    FROM EMP JOIN DEPT
    USING(DEPTNO)
    WHERE JOB='CLERK';

10. 

    SELECT ENAME, SAL, MGR, DNAME
    FROM EMP E JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO
    WHERE SAL>2000;

11. 

    SELECT EMPNO, ENAME, JOB, E.DEPTNO, DNAME, LOC
    FROM EMP E LEFT OUTER JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO;

12. 

    SELECT ENAME, DNAME
    FROM EMP E RIGHT OUTER JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO;

13. 

    SELECT EMPNO, DNAME, LOC
    FROM EMP E FULL OUTER JOIN DEPT D
    ON E.DEPTNO=D.DEPTNO;
